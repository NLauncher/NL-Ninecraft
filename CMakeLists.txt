cmake_minimum_required(VERSION 3.5...3.31)
project(ninecraft)

include_directories(${PROJECT_SOURCE_DIR}/stb)

SET(SDL_STATIC ON CACHE BOOL "Enable SDL_STATIC" FORCE)
SET(SDL_SHARED OFF CACHE BOOL "Disable SDL_SHARED" FORCE)
SET(SDL_MMX OFF CACHE BOOL "Disable SDL_MMX" FORCE)
SET(SDL_SSE OFF CACHE BOOL "Disable SDL_SSE" FORCE)
SET(SDL_SSE2 OFF CACHE BOOL "Disable SDL_SSE2" FORCE)
SET(SDL_SSE3 OFF CACHE BOOL "Disable SDL_SSE3" FORCE)
SET(SDL_SSEMATH OFF CACHE BOOL "Disable SDL_SSEMATH" FORCE)
SET(SDL_3DNOW OFF CACHE BOOL "Disable SDL_3DNOW" FORCE)
SET(SDL_TEST OFF CACHE BOOL "Disable SDL_TEST" FORCE)
SET(SDL_TESTS OFF CACHE BOOL "Disable SDL_TESTS" FORCE)
SET(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
SET(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
SET(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
SET(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)

if(EXISTS "${PROJECT_SOURCE_DIR}/SDL/src/audio/pulseaudio/SDL_pulseaudio.c")
    file(READ "${PROJECT_SOURCE_DIR}/SDL/src/audio/pulseaudio/SDL_pulseaudio.c" SDL_PULSE_C)
    if(NOT SDL_PULSE_C MATCHES "pulseaudio_candidates")
        string(REPLACE "pulseaudio_handle = SDL_LoadObject(pulseaudio_library);" "const char *pulseaudio_candidates[] = { pulseaudio_library, \"/usr/lib/libpulse.so.0\", \"/usr/lib/i386-linux-gnu/libpulse.so.0\", \"/usr/lib32/libpulse.so.0\", \"/app/lib/libpulse.so.0\", \"/app/lib/i386-linux-gnu/libpulse.so.0\", \"libpulse.so.0\", \"libpulse.so\", NULL }; for (int i = 0; pulseaudio_candidates[i] != NULL; ++i) { pulseaudio_handle = SDL_LoadObject(pulseaudio_candidates[i]); if (pulseaudio_handle) break; }" SDL_PULSE_C "${SDL_PULSE_C}")
        file(WRITE "${PROJECT_SOURCE_DIR}/SDL/src/audio/pulseaudio/SDL_pulseaudio.c" "${SDL_PULSE_C}")
    endif()
endif()
if(EXISTS "${PROJECT_SOURCE_DIR}/SDL/src/audio/alsa/SDL_alsa_audio.c")
    file(READ "${PROJECT_SOURCE_DIR}/SDL/src/audio/alsa/SDL_alsa_audio.c" SDL_ALSA_C)
    if(NOT SDL_ALSA_C MATCHES "alsa_candidates")
        string(REPLACE "alsa_handle = SDL_LoadObject(alsa_library);" "const char *alsa_candidates[] = { alsa_library, \"/usr/lib/libasound.so.2\", \"/usr/lib/i386-linux-gnu/libasound.so.2\", \"/usr/lib32/libasound.so.2\", \"/app/lib/libasound.so.2\", \"/app/lib/i386-linux-gnu/libasound.so.2\", \"libasound.so.2\", \"libasound.so\", NULL }; for (int i = 0; alsa_candidates[i] != NULL; ++i) { alsa_handle = SDL_LoadObject(alsa_candidates[i]); if (alsa_handle) break; }" SDL_ALSA_C "${SDL_ALSA_C}")
        file(WRITE "${PROJECT_SOURCE_DIR}/SDL/src/audio/alsa/SDL_alsa_audio.c" "${SDL_ALSA_C}")
    endif()
endif()

add_subdirectory(glad/cmake)
glad_add_library(glad REPRODUCIBLE LOADER API egl=1.5 gl:compatibility=2.0)
add_subdirectory(zlib)

add_subdirectory(anjni)
